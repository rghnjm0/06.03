def handle_room_availability(req):
    """Узнать свободные номера на дату."""
    p = req.get("payload", {}) or {}
    date_str = p.get("date") or datetime.now().strftime("%Y-%m-%d")

    # ТУТ — его логика запроса к БД гостиницы.
    # Пример-заглушка:
    return {
        "status": "ok",
        "date": date_str,
        "free_rooms": 5,               # сколько всего свободно
        "available_types": ["standard", "lux"]  # какие типы есть
    }
2. Функция бронирования номера:

python
def handle_book_room(req):
    """Забронировать конкретный номер."""
    p = req.get("payload", {}) or {}
    guest_name = p.get("guest_name", "Неизвестный гость")
    room_type = p.get("room_type", "standard")
    date_str = p.get("date", "")

    if not date_str:
        return {"status": "error", "error": "Не указана дата заезда"}

    # ТУТ — его логика записи в БД гостиницы.
    # Пример-заглушка:
    room_number = "204"  # выдали гостю номер
    
    return {
        "status": "ok",
        "booked": True,
        "room_number": room_number,
        "room_type": room_type,
        "guest_name": guest_name,
        "booking_id": f"ROOM-{int(time.time())}"
    }
3. Регистрация в словаре (у него в файле):

python

Как это будет работать
Ты заходишь в раздел «Интеграции».

Выбираешь Сервис: Гостиница.

Выбираешь Тип: Свободные номера.

В JSON автоматически подставится {"date": "2026-10-21"}. Нажимаешь «Отправить».

Через секунду обновляешь страницу, в таблице «Исходящие запросы» появляется ответ: {"status": "ok", "free_rooms": 5, "available_types": ["standard", "lux"]}.

То же самое с бронированием.
