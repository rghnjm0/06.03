private void AddNewModuleButtons()
{
    Panel side = null;
    foreach (Control c in Controls)
    {
        if (c is Panel p && p.Name == "panel1")
        {
            side = p;
            break;
        }
    }
    if (side == null) return;

    side.AutoScroll = true; // ползунок оставляем

    ExpUI.dll.ExpUIButton reportBtn = null;
    int maxBtnBottom = 0;
    foreach (Control c in side.Controls)
    {
        if (c is ExpUI.dll.ExpUIButton b && b.Width >= 180)
        {
            int bottom = b.Location.Y + b.Height;
            if (bottom > maxBtnBottom) maxBtnBottom = bottom;

            if (b.Text != null && b.Text.IndexOf("Отчет", StringComparison.OrdinalIgnoreCase) >= 0)
                reportBtn = b;
        }
    }

    const int gap = 10;
    const int btnH = 35;
    int y = reportBtn != null
        ? reportBtn.Location.Y + reportBtn.Height + gap
        : maxBtnBottom + gap;

    // если место занято — сдвигаем ниже, пока не будет свободно
    bool occupied;
    do
    {
        occupied = false;
        foreach (Control c in side.Controls)
        {
            if (!(c is ExpUI.dll.ExpUIButton) || c.Width < 180) continue;
            if (y < c.Location.Y + c.Height && y + btnH > c.Location.Y)
            {
                y = c.Location.Y + c.Height + gap;
                occupied = true;
                break;
            }
        }
    } while (occupied);

    ExpUI.dll.ExpUIButton btnServer = CreateSideMenuButton("Серверная", y);
    btnServer.Click += (s, ev) =>
    {
        Form f = new ServerAccessLogForm();
        f.Show();
        f.FormClosed += new FormClosedEventHandler(MainForm_FormClosed);
        Hide();
    };
    side.Controls.Add(btnServer);
    btnServer.BringToFront();
}

private ExpUI.dll.ExpUIButton CreateSideMenuButton(string text, int y)
{
    ExpUI.dll.ExpUIButton btn = new ExpUI.dll.ExpUIButton();
    btn.BackColor = Color.FromArgb(0, 116, 228);
    btn.BorderSize = 1;
    btn.ClickColor = Color.FromArgb(80, 255, 255, 255);
    btn.ColorTextForBorder = Color.White;
    btn.Cursor = Cursors.Hand;
    btn.DisableColor = Color.FromArgb(80, 255, 255, 255);
    btn.Font = new Font("Arial", 9.75F, FontStyle.Bold);
    btn.ForeColor = Color.White;
    btn.HoverColor = Color.FromArgb(25, 0, 0, 0);
    btn.Location = new Point(0, y);
    btn.Size = new Size(200, 35);
    btn.StyleButton = ExpUI.dll.ExpUIButton.styleButton.Standart;
    btn.Text = text;
    btn.TextAlign = ExpUI.dll.ExpUIButton.textAlign.MiddleRight;
    btn.ResponsiveText = true;
    btn.ResponsiveFontOn = new Font("Arial", 12F, FontStyle.Bold);
    return btn;
}
